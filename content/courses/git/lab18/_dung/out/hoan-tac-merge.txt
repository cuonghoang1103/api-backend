# Một commit merge có HAI cha — cha số 1 là phía main, cha số 2 là phía nhánh
[An] $ git log --merges -1 "--format=merge %h  parents: %p"
merge 5a123a1  parents: 77fb638 a89b8c2
[An] $ git show --no-patch "--format=parent 1 = %h %s" HEAD^1
parent 1 = 77fb638 Khoi tao du an NetBeans: Login, Home, web.xml
[An] $ git show --no-patch "--format=parent 2 = %h %s" HEAD^2
parent 2 = a89b8c2 Them TrangChuServlet (map /home)
