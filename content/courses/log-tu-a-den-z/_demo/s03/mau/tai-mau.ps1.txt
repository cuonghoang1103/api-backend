$loi = 0
foreach ($d in $ds) {
  $tep = Join-Path $Ra $d[0]
  if (-not (Test-Path -LiteralPath $tep)) { Invoke-WebRequest -Uri $d[1] -OutFile $tep -UseBasicParsing }
  $h = (Get-FileHash -LiteralPath $tep -Algorithm SHA256).Hash
  if ($h -eq $d[2]) { "OK   {0,10:N0}  {1}" -f (Get-Item -LiteralPath $tep).Length, $d[0] } else { "SAI  {0}  (hash {1})" -f $d[0], $h; $loi++ }
}
