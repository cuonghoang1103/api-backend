# Ch11 · 11.4 — bảng kiểm theo TẦNG, mỗi tầng một lệnh: DNS → routing → port → TLS → HTTP → app.
$ echo "== DNS (name -> IP; use dig for public names) =="
== DNS (name -> IP; use dig for public names) ==
$ getent hosts router

$ echo "== Routing (which gateway) =="
== Routing (which gateway) ==
$ ip route get 10.10.2.10 | head -1
10.10.2.10 via 10.10.1.254 dev eth0 src 10.10.1.10 uid 0 
$ echo "== Port (open or refused) =="
== Port (open or refused) ==
$ nc -zv 10.10.2.10 80 2>&1; nc -zv -w1 10.10.2.10 81 2>&1

nc: connect to 10.10.2.10 port 81 (tcp) failed: Connection refused
$ echo "== TLS (certificate + validity) =="
== TLS (certificate + validity) ==

subject=CN=labflow.test
notBefore=Nov  4 09:10:29 2026 GMT
notAfter=Nov  6 09:10:29 2026 GMT
