# đổi sang FULL tunnel: AllowedIPs = 0.0.0.0/0 — MỌI gói vào hầm
$ wg-quick down wg0 2>/dev/null; sed -i 's|^AllowedIPs = 10.66.0.0/24|AllowedIPs = 0.0.0.0/0|' /etc/wireguard/wg0.conf
$ wg-quick up wg0
