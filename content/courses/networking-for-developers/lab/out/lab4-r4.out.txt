== không đặt timeout (nhân tự bỏ cuộc) ==
$ t0=$(date +%s); curl -s -o /dev/null -w "  exit=%{exitcode}\n" http://10.10.2.99/ ; echo "  mất $(( $(date +%s) - t0 )) giây"
  exit=28
