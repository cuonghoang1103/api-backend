# Drill K1 — disk arm scheduling, cylinders 0-199, head at 80 moving up ||| Bài K1 — lập lịch tay đĩa, trụ 0-199, đầu đọc ở 80 đang đi lên
$ python3 osgcalc.py disk fcfs 80 "45 132 12 98 181 67 140 30"
== FCFS, head at 80 ==
order: 80 -> 45 -> 132 -> 12 -> 98 -> 181 -> 67 -> 140 -> 30
moves: 35 + 87 + 120 + 86 + 83 + 114 + 73 + 110 = 708 cylinders
$ python3 osgcalc.py disk sstf 80 "45 132 12 98 181 67 140 30" ms=2
== SSTF (shortest seek first), head at 80 ==
order: 80 -> 67 -> 45 -> 30 -> 12 -> 98 -> 132 -> 140 -> 181
moves: 13 + 22 + 15 + 18 + 86 + 34 + 8 + 41 = 237 cylinders
seek time = 237 x 2 ms = 474 ms
$ python3 osgcalc.py disk scan 80 "45 132 12 98 181 67 140 30"
== SCAN (elevator, to the edge), head at 80, moving up ==
order: 80 -> 98 -> 132 -> 140 -> 181 -> 199 -> 67 -> 45 -> 30 -> 12
moves: 18 + 34 + 8 + 41 + 18 + 132 + 22 + 15 + 18 = 306 cylinders
$ python3 osgcalc.py disk look 80 "45 132 12 98 181 67 140 30"
== LOOK (elevator, turn at last request), head at 80, moving up ==
order: 80 -> 98 -> 132 -> 140 -> 181 -> 67 -> 45 -> 30 -> 12
moves: 18 + 34 + 8 + 41 + 114 + 22 + 15 + 18 = 270 cylinders
$ python3 osgcalc.py disk cscan 80 "45 132 12 98 181 67 140 30"
== C-SCAN (to the edge, jump back to 0), head at 80, moving up ==
order: 80 -> 98 -> 132 -> 140 -> 181 -> 199 -> 0 -> 12 -> 30 -> 45 -> 67
moves: 18 + 34 + 8 + 41 + 18 + 199 + 12 + 18 + 15 + 22 = 385 cylinders
  (the return jump from the edge to 0 is counted here; some books leave it out - say which rule you use)
$ python3 osgcalc.py disk clook 80 "45 132 12 98 181 67 140 30"
== C-LOOK (jump back to lowest request), head at 80, moving up ==
order: 80 -> 98 -> 132 -> 140 -> 181 -> 12 -> 30 -> 45 -> 67
moves: 18 + 34 + 8 + 41 + 169 + 18 + 15 + 22 = 325 cylinders
$ python3 osgcalc.py disk look 80 "45 132 12 98 181 67 140 30" down
== LOOK (elevator, turn at last request), head at 80, moving down ==
order: 80 -> 67 -> 45 -> 30 -> 12 -> 98 -> 132 -> 140 -> 181
