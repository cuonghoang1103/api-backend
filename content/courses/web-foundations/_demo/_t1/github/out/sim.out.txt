### 1. 'Create repository' on the host = an empty bare repository
$ git init --bare -q --initial-branch=main github.com/an/ct-shop.git

$ ls github.com/an/ct-shop.git
config
description
HEAD
hooks
info
objects
refs

### 2. An: a local project with one commit, then connect it to the remote
$ git status --short --ignored
!! .env

$ git check-ignore -v .env
.gitignore:3:.env	.env

$ git remote add origin github.com/an/ct-shop.git

$ git remote -v
origin	github.com/an/ct-shop.git (fetch)
origin	github.com/an/ct-shop.git (push)

$ git push -u origin main
To github.com/an/ct-shop.git
 * [new branch]      main -> main
branch 'main' set up to track 'origin/main'.

$ git status -sb
## main...origin/main

### 6. A pull request is a branch on the remote, waiting to be merged
$ git switch -c feature/cart
Switched to a new branch 'feature/cart'

$ git push -u origin feature/cart
To github.com/an/ct-shop.git
 * [new branch]      feature/cart -> feature/cart
branch 'feature/cart' set up to track 'origin/feature/cart'.

$ git branch -a
* feature/cart
  main
  remotes/origin/HEAD -> origin/main
  remotes/origin/feature/cart
  remotes/origin/main

$ git ls-remote --heads origin
12c727d9c69294252e0e45a69990f1d54ae96f0c	refs/heads/feature/cart
4d21e2a00d865d078c3989a1a4d1e558879a211d	refs/heads/main
