sqlite> SELECT count(*) AS posts, sum(views) AS total_views, round(avg(views), 1) AS avg_views FROM posts WHERE published = 1;
+-------+-------------+-----------+
| posts | total_views | avg_views |
+-------+-------------+-----------+
|     4 |         545 |     136.3 |
+-------+-------------+-----------+

sqlite> SELECT user_id, count(*) AS posts, sum(views) AS views FROM posts GROUP BY user_id ORDER BY views DESC;
+---------+-------+-------+
| user_id | posts | views |
+---------+-------+-------+
|       2 |     2 |   303 |
|       1 |     3 |   245 |
|         |     1 |     0 |
+---------+-------+-------+
